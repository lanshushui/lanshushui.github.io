# 不要返回本地变量的引用（可以返回，但绝对不要）

有一种常见的 C++ 编程错误，是在函数里返回一个本地对象的引用。由于在函数结束时本地对象即被销毁，返回一个指向本地对象的引用属于未定义行为。

在 C++11 之前，返回一个本地对象意味着这个对象会被拷贝，除非编译器发现可以做返回值优化（named return value optimization，或 NRVO），能把对象直接构造到调用者的栈上。

从 C++11 开始，返回值优化仍可以发生，但在没有返回值优化的情况下，编译器将试图把本地对象移动出去，而不是拷贝出去。这一行为不需要程序员手工用 std::move 进行干预 —— 使用 std::move 对于移动行为没有帮助，反而会影响返回值优化。

#include <iostream>
#include <memory>

using namespace std;

class C
{
public:
    C()noexcept { std::cout << "C()" << std::endl; }
    C(const C&)noexcept { std::cout << "C(const C&)" << std::endl; }
    C(C&&)noexcept { std::cout << "C(C&&)" << std::endl; }
};

C fun1()
{
    C c;
    // 简单返回对象一般有 NRVO
    return c;
}

C fun2()
{
    C c;
    //std::move禁止 NRVO
    return std::move(c);
}

C fun3(bool f)
{
    C c1, c2;
    //有分支一般无 NRVO
    return f ? c1 : c2;
}

int main(int argc, char** argv)
{
    std::cout << "fun1" << std::endl;
    C c1 = fun1();
    std::cout << "fun2" << std::endl;
    C c2 = fun2();
    std::cout << "fun3" << std::endl;
    C c3 = fun3(true);
}
/*
fun1
C()
fun2
C()
C(C&&)
fun3
C()
C()
C(const C&)
*/

# 测试场景

# 无 NRVO 场景下返回一个临时变量会进行复制拷贝，不管是否有 const 引用接受这个临时变量

看网上的文档。说用普通变量去接受函数返回值，无 NRVO 情况下会导致复制操作，可以用 const 引用去延迟临时变量的生命周期，避免复制操作的产生，但实际实验下来 发现没有用，没有变量接住临时变量都会进行复制操作

#include <iostream>
#include <memory>

using namespace std;

class C
{
    public:
    C()noexcept { std::cout << "C()" << std::endl; }
    C(const C&)noexcept { std::cout << "C(const C&)" << std::endl; }
    C(C&&)noexcept { std::cout << "C(C&&)" << std::endl; }

    void eat() const{
        std::cout << "eat" << std::endl;
    }
    ~C(){
        std::cout << "~C()" << std::endl;
    }
};

C fun3(bool f)
{
    C c1, c2;
    //有分支一般无 NRVO
    return f ? c1 : c2;
}

int main(int argc, char** argv)
{
    std::cout << "----test1----" << std::endl;
    {
        fun3(true);
        /**
        ----test1----
            C()
            C()
            C(const C&)
            ~C()
            ~C()
            ~C()
			没有变量接住临时变量都会进行复制操作
        **/
    }

    std::cout << "----test2----" << std::endl;
    {
        C c = fun3(true);
        std::cout << "-----" << std::endl;
        c.eat();
         /**
            ----test2----
            C()
            C()
            C(const C&)
            ~C()
            ~C()
            -----
            eat
            ~C()
			普通变量接住函数返回的临时变量，会产生复制操作
        **/ 
        
    }
    std::cout << "----test3----" << std::endl;
    {
        const C& c = fun3(true);
        std::cout << "-----" << std::endl;
        c.eat();
        /**
        ----test3----
            C()
            C()
            C(const C&)
            ~C()
            ~C()
            -----
            eat
            ~C()
            const引用接住函数返回的临时变量，也会产生复制操作，没看到延迟临时变量的生命周期
        **/
    }
}
# 当引用变量去接受函数返回的引用时，才不会发生复制操作
#include <iostream>

using namespace std;

class C
{
public:
    C()noexcept { std::cout << "C()" << std::endl; }
    C(const C&)noexcept { std::cout << "C(const C&)" << std::endl; }
    C(C&&)noexcept { std::cout << "C(C&&)" << std::endl; }

    void eat() const{
        std::cout << "eat" << std::endl;
    }
    ~C(){
        std::cout << "~C()" << std::endl;
    }
};
class A{
public:
   C  getC(){
        return c;
    }
    C& getRefC(){
        return  c;
   }
private:
    C c;
};

int main(int argc, char** argv)
{
    std::cout << "----test1----" << std::endl;
    {
         A a;
        std::cout << "----after init----" << std::endl;
         C c2 =a.getC();
        /**
        ----test1----
        C()
        ----after init----
        C(const C&)
        ~C()
        ~C()
        普通变量接受临时变量 发生复制
        **/
    }
    std::cout << "----test2----" << std::endl;
    {
        A a;
        std::cout << "----after init----" << std::endl;
        const C& c2 =a.getC();
        /**
        ----test2----
        C()
        ----after init----
        C(const C&)
        ~C()
        ~C()
        const引用接受临时变量 发生复制
        **/
    }
    std::cout << "----test3----" << std::endl;
    {
        A a;
        std::cout << "----after init----" << std::endl;
        const C c2 =a.getRefC();
        /**
        ----test3----
        C()
        ----after init----
        C(const C&)
        ~C()
        ~C()
        普通变量接受返回的引用 发生复制
        **/        
    }
    std::cout << "----test4----" << std::endl;
    {
        A a;
        std::cout << "----after init----" << std::endl;
        const C& c2 =a.getRefC();
        /**
        ----test4----
        C()
        ----after init----
        ~C()
        const引用接受返回的引用 不会发生复制
        **/              
    }
}
